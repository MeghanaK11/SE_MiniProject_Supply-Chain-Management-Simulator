1"""CSV export and text summary (SCM-F-034, SCM-SR-003).""" 

2import csv 

3import os

5COLUMNS = ("day", "product_id", "demand", "shipped", "stock_end", "backorder", "on_order", "forecast") 

6FORMULA_PREFIXES = ("=", "+", "-", "@") 

7SUMMARY_KPI_KEYS = ("fill_rate", "service_level", "avg_inventory_units", 

8 "inventory_turnover", "total_backorder_end", "forecast_mae") 

9SUMMARY_COST_KEYS = ("holding", "stockout", "ordering", "purchase", "transport", "total")

12def _safe_cell(value): 

13 """Prefix text that a spreadsheet could read as a formula with an apostrophe; leave numbers alone.""" 

14 if isinstance(value, str) and value.startswith(FORMULA_PREFIXES): 

15 return "'" + value 

16 return value

19def export_daily_csv(daily_records: list, path: str) -> None: 

20 """Write one row per product per day to path, creating parent folders.""" 

21 for index, record in enumerate(daily_records): 

22 for column in COLUMNS: 

23 if column not in record: 

24 raise ValueError(f"record {index} is missing column '{column}'") 

25 folder = os.path.dirname(path) 

26 if folder: 

27 os.makedirs(folder, exist_ok=True) 

28 with open(path, "w", newline="", encoding="utf-8") as handle: 

29 writer = csv.writer(handle) 

30 writer.writerow(COLUMNS) 

31 for record in daily_records: 

32 writer.writerow([_safe_cell(record[column]) for column in COLUMNS])

35def format_summary(kpis: dict, costs: dict) -> str: 

36 """Printable summary of costs (Rs, 2 decimals) and KPIs (rates in percent, 2 decimals).""" 

37 for key in SUMMARY_KPI_KEYS: 

38 if key not in kpis: 

39 raise ValueError(f"kpis is missing '{key}'") 

40 for key in SUMMARY_COST_KEYS: 

41 if key not in costs: 

42 raise ValueError(f"costs is missing '{key}'") 

43 lines = [ 

44 "Costs (Rs)", 

45 f" Holding cost: {costs['holding']:.2f}", 

46 f" Stockout cost: {costs['stockout']:.2f}", 

47 f" Ordering cost: {costs['ordering']:.2f}", 

48 f" Purchase cost: {costs['purchase']:.2f}", 

49 f" Transport cost: {costs['transport']:.2f}", 

50 f" Total cost: {costs['total']:.2f}", 

51 "Service", 

52 f" Fill rate: {kpis['fill_rate'] * 100:.2f}%", 

53 f" Service level: {kpis['service_level'] * 100:.2f}%", 

54 f" Backorder units at end: {kpis['total_backorder_end']}", 

55 "Inventory", 

56 f" Average inventory units: {kpis['avg_inventory_units']:.2f}", 

57 f" Inventory turnover: {kpis['inventory_turnover']:.2f}", 

58 "Forecast", 

59 f" Mean absolute error: {kpis['forecast_mae']:.2f}", 

60 ]
